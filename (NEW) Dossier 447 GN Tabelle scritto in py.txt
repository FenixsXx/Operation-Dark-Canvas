def wrap_text(text, width):
    """Ritorna una lista di righe di lunghezza massima width."""
    import textwrap
    return textwrap.wrap(text, width=width)

def print_table_3col(headers, rows, widths):
    """
    Stampa una tabella a 3 colonne con word-wrap.
    widths: tuple di 3 interi (larghezza caratteri per colonna).
    """
    h1, h2, h3 = headers
    w1, w2, w3 = widths
    sep = f"+{'-'*(w1+2)}+{'-'*(w2+2)}+{'-'*(w3+2)}+"

    def fmt_cell(text, width):
        lines = wrap_text(text, width)
        return [f" {line:<{width}} " for line in lines] if lines else [f" {'':<{width}} "]

    def print_row(c1, c2, c3):
        l1 = fmt_cell(c1, w1)
        l2 = fmt_cell(c2, w2)
        l3 = fmt_cell(c3, w3)
        max_lines = max(len(l1), len(l2), len(l3))
        for i in range(max_lines):
            a = l1[i] if i < len(l1) else f" {'':<{w1}} "
            b = l2[i] if i < len(l2) else f" {'':<{w2}} "
            c = l3[i] if i < len(l3) else f" {'':<{w3}} "
            print(f"|{a}|{b}|{c}|")

    print(sep)
    print(f"| {h1:<{w1}} | {h2:<{w2}} | {h3:<{w3}} |")
    print(sep)
    for r in rows:
        print_row(r[0], r[1], r[2])
        print(sep)
    print()
